# Exemplo de código demonstrando o uso dos métodos find() e len()

# Definindo uma string de exemplo

frase = "Olá, bem-vindo ao mundo da programação com Python!"

# Usando o método find() para encontrar a posição da palavra "programação"

posicao = frase.find("programação")

print("A palavra 'programação' está na posição:", posicao)

# Usando o método len() para encontrar o comprimento da frase

comprimento = len(frase)

print("O comprimento da frase é:", comprimento)

# Verificação adicional usando find() para um exemplo prático

palavra = "Python"

if frase.find(palavra) != -1:

print(f"A palavra '{palavra}' foi encontrada na frase.")

else:

print(f"A palavra '{palavra}' não foi encontrada na frase.")
